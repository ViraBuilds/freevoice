Drop-in OpenAI SDK Compatibility

Point any OpenAI TTS client to http://127.0.0.1:8000/v1. No API key validation is enforced. 

from openai import OpenAI

client = OpenAI(
    base_url="http://127.0.0.1:8000/v1",
    api_key="local-freevoice",
)

with client.audio.speech.with_streaming_response.create(
    model="kokoro-82m",
    voice="af_heart",
    input="Hello from FreeVoice!",
    response_format="mp3",
) as response:
    response.stream_to_file("output.mp3")
